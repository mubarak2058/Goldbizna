let items = JSON.parse(localStorage.getItem("goldbizna_inventory") || "[]"); function save() { localStorage.setItem("goldbizna_inventory", JSON.stringify(items)); render(); } function addItem() { const n = name.value.trim(); if (!n) { alert("Enter product"); return; } items.push({ n: n, c: cat.value, b: buy.value, s: sell.value, q: qty.value }); name.value = ""; cat.value = ""; buy.value = ""; sell.value = ""; qty.value = ""; save(); } function editItem(i) { const item = items[i]; name.value = item.n; cat.value = item.c; buy.value = item.b; sell.value = item.s; qty.value = item.q; items.splice(i, 1); save(); } function del(i) { if (confirm("Delete this product?")) { items.splice(i, 1); save(); } } function render() { tb.innerHTML = ""; items.forEach((x, i) => { tb.innerHTML += ` ${x.n} ${x.c} ${x.b} ${x.s} ${x.q} `; }); } render();
